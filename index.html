const applications = files .filter(file => file.type === "file" && (file.name.toLowerCase().endsWith(".html") || file.name.toLowerCase().endsWith(".pdf")) && file.name.toLowerCase() !== "index.html" ) .sort((a, b) => a.name.localeCompare(b.name) ); if (applications.length === 0) { container.textContent = "No applications found."; return; } const list = document.createElement("ul"); applications.forEach(app => { const item = document.createElement("li"); const link = document.createElement("a"); link.href = app.name; link.textContent = app.name.replace(/\.(html|pdf)$/i, ""); item.appendChild(link); list.appendChild(item); });
